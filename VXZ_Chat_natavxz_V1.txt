local SAVE_KEY = "VXZChat_natavxz_V1"

local function SaveData()
    -- En executor, usamos setclipboard o writefile si disponible
    -- Fallback seguro con pcall
    pcall(function()
        if writefile then
            local data = {
                chatWindows = {},
                floatConfig = State.floatConfig,
                textboxes   = {},
                floatLocked = State.floatLocked,
            }
            for i, cw in ipairs(State.chatWindows) do
                data.chatWindows[i] = {
                    title    = cw.title,
                    symStart = cw.symStart,
                    symEnd   = cw.symEnd,
                    extraBtn = cw.extraBtn,
                    extraAuto= cw.extraAuto,
                }
            end
            for i, tb in ipairs(State.textboxes) do
                data.textboxes[i] = {
                    title    = tb.title,
                    symStart = tb.symStart,
                    symEnd   = tb.symEnd,
                    color    = {tb.color.R, tb.color.G, tb.color.B},
                    btnColor = {tb.btnColor.R, tb.btnColor.G, tb.btnColor.B},
                    size     = tb.size,
                    clearOnSend = tb.clearOnSend,
                }
            end
            writefile(SAVE_KEY..".json", game:GetService("HttpService"):JSONEncode(data))
        end
    end)
end

local function LoadData()
    local ok, result = pcall(function()
        if readfile and isfile and isfile(SAVE_KEY..".json") then
            local raw  = readfile(SAVE_KEY..".json")
            return game:GetService("HttpService"):JSONDecode(raw)
        end
    end)
    if ok and result then
        return result
    end
    return nil
end

local function SendChat(message)
    pcall(function()
        if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
            TextChatService.TextChannels.RBXGeneral:SendAsync(message)
        else
            game:GetService("ReplicatedStorage")
                .DefaultChatSystemChatEvents
                .SayMessageRequest
                :FireServer(message, "All")
        end
    end)
end

local function Tween(obj, props, t, style, dir)
    local ti = TweenInfo.new(
        t or 0.25,
        style or Enum.EasingStyle.Quart,
        dir   or Enum.EasingDirection.Out
    )
    TweenService:Create(obj, ti, props):Play()
end

local function AddStroke(obj, color, thickness, trans)
    local s = Instance.new("UIStroke")
    s.Color        = color or C.BORDER
    s.Thickness    = thickness or 1.2
    s.Transparency = trans or 0.4
    s.Parent       = obj
    return s
end

local function AddCorner(obj, radius)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius or 8)
    c.Parent       = obj
    return c
end

    p.PaddingRight  = UDim.new(0, px or 8)
    p.PaddingTop    = UDim.new(0, py or 6)
    p.PaddingBottom = UDim.new(0, py or 6)
    p.Parent        = obj
    return p
end

-- FadeIn / FadeOut para frames
local function FadeIn(frame, duration)
    frame.Visible = true
    frame.BackgroundTransparency = 1
    Tween(frame, {BackgroundTransparency = 0.08}, duration or 0.2)
end

local function FadeOut(frame, duration, callback)
    Tween(frame, {BackgroundTransparency = 1}, duration or 0.18)
    task.delay(duration or 0.18, function()
        frame.Visible = false
        if callback then callback() end
    end)
end

-- Hacer Frame arrastrable (drag)
local function MakeDraggable(frame, handle, lockRef)
    -- lockRef: función que retorna true si está bloqueado
    local dragging, dragStart, startPos
    local function Update(input)
        local delta = input.Position - dragStart
        local newPos = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
        frame.Position = newPos
    end
    handle.InputBegan:Connect(function(input)
        if lockRef and lockRef() then return end
        if input.UserInputType == Enum.UserInputType.Touch
        or input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging  = true
            dragStart = input.Position
            startPos  = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    handle.InputChanged:Connect(function(input)
        if dragging and (
            input.UserInputType == Enum.UserInputType.Touch or
            input.UserInputType == Enum.UserInputType.MouseMovement
        ) then
            Update(input)
        end
    end)
end

local dynamicTBList = {}  -- referencia a los frames creados

local function CreateDynamicTextBox(cfg)
    if #dynamicTBList >= 5 then
        tbCountLbl.Text = "TextBoxes creados: 5 / 5  (máximo alcanzado)"
        tbCountLbl.TextColor3 = C.BTN_CLOSE
        return
    end

    local idx = #dynamicTBList + 1
    local w   = math.clamp(cfg.w or 220, 140, 300)
    local h   = math.clamp(cfg.h or 180, 120, 260)

    local dtFrame = MakeFrame(ScreenGui,
        UDim2.new(0,w,0,h),
        UDim2.new(0, 50 + (idx-1)*16, 0, 100 + (idx-1)*16),
        C.BG_CARD, 0.06
    )
    dtFrame.Name  = "DynamicTB_"..idx
    dtFrame.ZIndex = 9
    AddCorner(dtFrame, 10)
    AddStroke(dtFrame, C.ACCENT2, 1.2, 0.35)

    -- Header arrastrable
    local dtHeader = MakeFrame(dtFrame,
        UDim2.new(1,0,0,30),
        UDim2.new(0,0,0,0),
        C.BG_DARK, 0.0
    )
    AddCorner(dtHeader, 10)
    local dtAccent = Instance.new("Frame")
    dtAccent.Size             = UDim2.new(0,3,1,0)
    dtAccent.BackgroundColor3 = C.ACCENT2
    dtAccent.BorderSizePixel  = 0
    dtAccent.Parent           = dtHeader

    local dtTitle = MakeLabel(dtHeader, "  " .. (cfg.title or "TextBox"),
        UDim2.new(1,-36,1,0), UDim2.new(0,4,0,0),
        C.ACCENT2, 11, Enum.Font.GothamBold)

    local dtClose = MakeButton(dtHeader, "✕",
        UDim2.new(0,22,0,20), UDim2.new(1,-24,0,5),
        C.BTN_CLOSE, C.WHITE, 11
    )

    -- Lock
    local dtLock = Instance.new("TextLabel")
    dtLock.Size             = UDim2.new(0,16,0,16)
    dtLock.Position         = UDim2.new(1,-40,0,7)
    dtLock.Text             = "🔓"
    dtLock.TextSize         = 11
    dtLock.BackgroundColor3 = C.BG_MID
    dtLock.TextColor3       = C.TEXT_DIM
    dtLock.Font             = Enum.Font.Gotham
    dtLock.BackgroundTransparency = 0.2
    dtLock.ZIndex = 12
    dtLock.Parent = dtHeader
    AddCorner(dtLock, 4)

    -- Sym label
    local dtSymLbl = MakeLabel(dtFrame, "",
        UDim2.new(1,-8,0,12), UDim2.new(0,4,0,32),
        C.TEXT_DIM, 9, Enum.Font.Gotham)
    local s = cfg.symStart ~= "ninguno" and cfg.symStart or ""
    local e = cfg.symEnd   ~= "ninguno" and cfg.symEnd   or ""
    dtSymLbl.Text = "  ["..s.."] msg ["..e.."]"

    -- TextBox
    local dtBox = MakeTextBox(dtFrame,
        "Escribe aquí...",
        UDim2.new(1,-8,1,-82), UDim2.new(0,4,0,46)
    )
    dtBox.MultiLine   = true
    dtBox.TextWrapped = true

    -- Enviar
    local dtSend = MakeButton(dtFrame, "▶ ENVIAR",
        UDim2.new(1,-8,0,26), UDim2.new(0,4,1,-32),
        C.BTN_SEND, C.WHITE, 12
    )

    -- Eliminar
    local dtDel = MakeButton(dtFrame, "🗑",
        UDim2.new(0,26,0,26), UDim2.new(1,-30,1,-32),
        C.BTN_CLOSE, C.WHITE, 13
    )

    -- Lock drag
    local dtLocked = false
    dtLock.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch
        or input.UserInputType == Enum.UserInputType.MouseButton1 then
            dtLocked = not dtLocked
            dtLock.Text = dtLocked and "🔒" or "🔓"
        end
    end)

    MakeDraggable(dtFrame, dtHeader, function() return dtLocked end)

    dtSend.MouseButton1Click:Connect(function()
        local raw = dtBox.Text
        if raw == "" then return end
        local ss = cfg.symStart ~= "ninguno" and cfg.symStart or ""
        local se = cfg.symEnd   ~= "ninguno" and cfg.symEnd   or ""
        SendChat(ss .. raw .. se)
        if cfg.clearOnSend then dtBox.Text = "" end
    end)

    dtClose.MouseButton1Click:Connect(function()
        Tween(dtFrame, {BackgroundTransparency = 1}, 0.18)
        task.delay(0.2, function() dtFrame.Visible = false end)
    end)

    dtDel.MouseButton1Click:Connect(function()
        -- Eliminar de la lista
        for k, v in ipairs(dynamicTBList) do
            if v.frame == dtFrame then
                table.remove(dynamicTBList, k)
                break
            end
        end
        dtFrame:Destroy()
        tbCountLbl.Text = "TextBoxes creados: " .. #dynamicTBList .. " / 5"
        tbCountLbl.TextColor3 = C.TEXT_DIM
        SaveData()
    end)

    -- FadeIn
    dtFrame.BackgroundTransparency = 1
    Tween(dtFrame, {BackgroundTransparency = 0.06}, 0.22)

    table.insert(dynamicTBList, {
        frame      = dtFrame,
        title      = cfg.title or "TextBox",
        symStart   = cfg.symStart or "ninguno",
        symEnd     = cfg.symEnd or "ninguno",
        clearOnSend= cfg.clearOnSend or false,
        w = w, h = h,
    })

    tbCountLbl.Text = "TextBoxes creados: " .. #dynamicTBList .. " / 5"
    tbCountLbl.TextColor3 = C.TEXT_DIM
    State.textboxes = dynamicTBList
    SaveData()
end

-- [15] CARGA INICIAL DE DATOS GUARDADOS
-- ═══════════════════════════════════════════════

task.spawn(function()
    local saved = LoadData()
    if not saved then return end

    -- Restaurar chats
    if saved.chatWindows then
        for i, cw in ipairs(saved.chatWindows) do
            if State.chatWindows[i] then
                State.chatWindows[i].title    = cw.title    or State.chatWindows[i].title
                State.chatWindows[i].symStart = cw.symStart or "ninguno"
                State.chatWindows[i].symEnd   = cw.symEnd   or "ninguno"
                State.chatWindows[i].extraBtn = cw.extraBtn or false
                State.chatWindows[i].extraAuto= cw.extraAuto or false
                -- Actualizar tarjeta
                local card = ScrollChats:FindFirstChild("ChatCard_"..i)
                if card then
                    local tl = card:FindFirstChild("TitleCard_"..i)
                    if tl then tl.Text = State.chatWindows[i].title end
                end
            end
        end
    end

    -- Restaurar configuración del botón flotante
    if saved.floatConfig then
        local fc = saved.floatConfig
        State.floatConfig = fc
        local sz = math.clamp(fc.size or 54, 30, 90)
        FloatHolder.Size = UDim2.new(0,sz,0,sz)
        FloatBtn.Text    = fc.emoji or "💬"
        FloatBtn.BackgroundTransparency = fc.alpha or 0
        if fc.posX and fc.posY then
            FloatHolder.Position = UDim2.new(0, fc.posX, 0, fc.posY)
        end
    end

    -- Restaurar candado
    if saved.floatLocked ~= nil then
        State.floatLocked = saved.floatLocked
        LockIcon.Text = State.floatLocked and "🔒" or "🔓"
    end

    -- Restaurar textboxes dinámicos
    if saved.textboxes then
        for _, tb in ipairs(saved.textboxes) do
            CreateDynamicTextBox({
                title      = tb.title,
                w          = tb.size and tb.size[1] or 220,
                h          = tb.size and tb.size[2] or 180,
                symStart   = tb.symStart or "ninguno",
                symEnd     = tb.symEnd   or "ninguno",
                clearOnSend= tb.clearOnSend or false,
            })
        end
    end
end)
